## 命令显示获取已安装包的信息（包名与版本号）
pip list | grep package_name 

## 命令能显示该安装的包的相关信息，其中包括它的安装路径。
pip show package_name

## pipreqs 生成 requirements.txt 虚拟环境依赖问题忽略.venv
pipreqs --encoding=utf8 --ignore .venv --force

## 启动博客系统命令
uvicorn rewrz.main:app --reload
uvicorn rewrz.main:app --reload --port 8080

## 单元测试命令
pytest .\tests\test_crud_post.py
$env:PYTHONPATH = $pwd.Path; pytest tests/test_crud_setting.py -v
